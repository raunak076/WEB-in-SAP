var cid = "";
function switchTab(id){
  var panels=["panel1","panel2","panel3"];
  for(var i=0;i<panels.length;i++){
    var p=document.getElementById(panels[i]);
    if(p) p.style.display=(panels[i]===id)?"block":"none";
